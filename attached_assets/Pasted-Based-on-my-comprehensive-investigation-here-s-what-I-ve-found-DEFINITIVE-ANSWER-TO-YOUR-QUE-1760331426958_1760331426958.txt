Option A: Two API Calls (More Reliable)
// Step 1: Get properties in radius
async getPropertiesWithSales(lat: number, lng: number, radius: number) {
  const propertiesResponse = await axios.get(
    'https://api.gateway.attomdata.com/propertyapi/v1.0.0/property/snapshot',
    {
      headers: { 'apikey': process.env.ATTOM_API_KEY },
      params: {
        latitude: lat,
        longitude: lng,
        radius: radius,
        pagesize: 100
      }
    }
  );

  const properties = propertiesResponse.data.property || [];
  
  // Step 2: Get sales history for each property
  const propertiesWithSales = await Promise.all(
    properties.map(async (prop) => {
      try {
        const salesResponse = await axios.get(
          'https://api.gateway.attomdata.com/propertyapi/v1.0.0/property/expandedprofile',
          {
            headers: { 'apikey': process.env.ATTOM_API_KEY },
            params: {
              address1: prop.address.line1,
              address2: `${prop.address.locality}, ${prop.address.countrySubd}`
            }
          }
        );
        
        return {
          ...prop,
          saleData: salesResponse.data.property[0].sale,
          saleHistory: salesResponse.data.property[0].saleTransHistory
        };
      } catch (error) {
        console.log(`No sale data for ${prop.address.oneLine}`);
        return { ...prop, saleData: null };
      }
    })
  );

  // Filter to only properties with recent sales
  return propertiesWithSales.filter(p => 
    p.saleData && p.saleData.saleRecDate
  );
}
